- 两面模型（主面 / 副面）
- 特性数据模型与主动特性结算

- 战斗钩子总线（含击杀者归属、跌破阈值）

## Task 2：两面模型（主面与副面）

**目的**：spec §2.1。每张字有一个主面，副面的本体数值按 `SideFacePercent`（默认 60）结算。

**只缩放连续量**：判定直接复用现有白名单 `BattleEngine.TakesElementPercent`（伤害、治疗、护盾等）。冻结回合、灼层数这类离散量不缩放。

**恒等**：`MainFace == null` 时完全不缩放；现有字表没有 `mainFace`，因此整体恒等。

**Files**：
- Create：`Core/CardFace.cs`（+ .meta）
- Modify：`Core/CharDef.cs`（构造函数末尾追加参数 `CardFace? mainFace = null`）
- Modify：`Core/BattleEngine.cs`
  - `BattleConfig` 加 `SideFacePercent`
  - 新增 `FaceOf`、`ApplySideFacePercent`
  - `ApplyEffects` 中接入副面缩放
- Modify：`Data/ConfigLoader.cs`（`CharDto.MainFace` 和解析）
- Test：`Tests/SideFaceTests.cs`（+ .meta）

**Interfaces**：
- Produces：
  - `public enum CardFace { Feature, Attack }`
  - `CharDef.MainFace : CardFace?`
  - `public static CardFace BattleEngine.FaceOf(CharDef def, bool attackMode)`
  - `public static int BattleEngine.ApplySideFacePercent(int value, CharDef def, bool attackMode, EffectKind kind, int sideFacePercent)`
  - `BattleConfig.SideFacePercent : int = 60`

  ```csharp
  using System.Collections.Generic;
  using System.Linq;

  namespace Brushblade.Core
  {
      /// <summary>特性解锁规则(spec v6 §1):等级达到槽位值即解锁;被更高槽位 Replaces 的不再生效;
      /// 结果按槽位升序 —— 即 R3 的 Lv1/Lv3 → Lv5/Lv8 顺序。</summary>
      public static class TraitRules
      {
          public static IReadOnlyList<TraitDef> Unlocked(CharDef def, int cardLevel)
          {
              var unlocked = def.Traits.Where(t => t.UnlockLevel <= cardLevel).ToList();
              var replaced = new HashSet<TraitSlot>(unlocked.Where(t => t.Replaces.HasValue).Select(t => t.Replaces.Value));
              return unlocked.Where(t => !replaced.Contains(t.Slot)).OrderBy(t => (int)t.Slot).ToList();
          }

          public static IReadOnlyList<TraitDef> ActiveTraits(CharDef def, CardFace face, int cardLevel) =>
              Unlocked(def, cardLevel).Where(t => t.Form == TraitForm.Active && t.AppliesTo(face)).ToList();
      }
  }
  ```

  ```csharp
  private static T ParseEnum<T>(string raw, T fallback, string charId, string what) where T : struct, Enum
  {
      if (string.IsNullOrEmpty(raw)) return fallback;
      if (!Enum.TryParse<T>(raw, out var v) || !Enum.IsDefined(typeof(T), v))
          throw new ConfigException($"字「{charId}」的{what}未知:{raw}");
      return v;
  }

  private static IReadOnlyList<TraitDef> ParseTraits(CharDto dto)
  {
      if (dto.Traits == null) return null;
      var traits = new List<TraitDef>();
      var slots = new HashSet<TraitSlot>();
      foreach (var t in dto.Traits)
      {
          if (string.IsNullOrEmpty(t.Slot))
              throw new ConfigException($"字「{dto.Id}」有特性缺少 slot");
          var slot = ParseEnum(t.Slot, TraitSlot.Lv1, dto.Id, "特性槽位");
          if (!slots.Add(slot))
              throw new ConfigException($"字「{dto.Id}」的特性槽位重复:{t.Slot}");
          var face = ParseEnum(t.Face, TraitFace.Both, dto.Id, "特性作用面");
          var form = ParseEnum(t.Form, TraitForm.Active, dto.Id, "特性形态");
          TraitSlot? replaces = string.IsNullOrEmpty(t.Replaces)
              ? (TraitSlot?)null
              : ParseEnum(t.Replaces, TraitSlot.Lv1, dto.Id, "特性替换槽位");
          traits.Add(new TraitDef(slot, face, form, replaces, t.Name ?? "", ParseEffects(dto, t.Effects ?? new List<EffectDto>())));
      }
      foreach (var t in traits)
          if (t.Replaces.HasValue && (!slots.Contains(t.Replaces.Value) || (int)t.Replaces.Value >= (int)t.Slot))
              throw new ConfigException($"字「{dto.Id}」的特性「{t.Name}」替换了不存在或更高的槽位:{t.Replaces}");
      return traits;
  }

  /// <summary>主动特性若需要(敌方/友方)目标,它所作用的面的本体也必须选同类目标 ——
  /// 否则出手时 targetIndex / allySlot 停在缺省值,特性悄悄空转(与 NeedsTarget 注释里 C1 那次同型)。</summary>
  private static void ValidateTraitTargets(CharDef def)
  {
      foreach (var t in def.Traits)
      {
          if (t.Form != TraitForm.Active) continue;
          foreach (var face in new[] { CardFace.Feature, CardFace.Attack })
          {
              if (!t.AppliesTo(face)) continue;
              bool attackMode = face == CardFace.Attack;
              if (t.Effects.Any(BattleEngine.EffectNeedsTarget) && !BattleEngine.NeedsTarget(def, attackMode))
                  throw new ConfigException($"字「{def.Id}」的特性「{t.Name}」需要敌方目标,但该面本体不选敌方目标");
              if (t.Effects.Any(BattleEngine.EffectNeedsAllyTarget) && !BattleEngine.NeedsAllyTarget(def, attackMode))
                  throw new ConfigException($"字「{def.Id}」的特性「{t.Name}」需要友方目标,但该面本体不选友方目标");
          }
      }
  }
  ```

- [ ] **Step 4：`BattleEngine` 总线骨架**
  ```csharp
  private readonly List<IBattleHookListener> _hookListeners = new List<IBattleHookListener>();

  /// <summary>触发深度(R4):特性运行时进入/退出触发结算时增减。不进快照 —— 只在一次结算内部非 0。</summary>
  public int TriggerDepth { get; private set; }
  internal void EnterTrigger() => TriggerDepth++;
  internal void ExitTrigger() => TriggerDepth--;

  internal void AddHookListener(IBattleHookListener listener) => _hookListeners.Add(listener);

  /// <summary>无监听器时直接返回(恒等)。遍历副本:监听器可在回调里增删监听器。</summary>
  private void Raise(HookKind kind, UnitRef subject, UnitRef other, int amount = 0,
      StatusKind status = default, string charId = null, EffectSource source = EffectSource.None)
  {
      if (_hookListeners.Count == 0) return;
      var args = new HookArgs(kind, subject, other, amount, status, charId, source, TriggerDepth);
      foreach (var listener in _hookListeners.ToArray()) listener.OnHook(this, args);
  }

  public const int HpThresholdPercent = 50;
  private bool _playerThresholdCrossed;
  private readonly HashSet<int> _summonThresholdCrossed = new HashSet<int>();

  /// <summary>R5:生命从 ≥50% 跌到 <50% 时触发,每单位每场一次;复活不清标记。</summary>
  private void CheckThreshold(UnitRef unit, int hpBefore, int hpAfter, int maxHp)
  {
      if (maxHp <= 0) return;
      bool crossed = hpBefore * 100 >= maxHp * HpThresholdPercent && hpAfter * 100 < maxHp * HpThresholdPercent;
      if (!crossed) return;
      if (unit.Side == UnitSide.Player)
      {
          if (_playerThresholdCrossed) return;
          _playerThresholdCrossed = true;
      }
      else if (unit.Side == UnitSide.Summon)
      {
          if (!_summonThresholdCrossed.Add(unit.Index)) return;
      }
      else return;
      Raise(HookKind.HpThresholdCrossed, unit, UnitRef.None, HpThresholdPercent);
  }
  ```

  ```csharp
  using System;
  using System.Collections.Generic;
  using System.Linq;

  namespace Brushblade.Core
  {
      /// <summary>跨场开局效果(spec v6 §5.1):之后 BattlesLeft 场,每场开局对全场(不选目标)结算一次。
      /// 用可写属性的 POCO 而不是 EffectDef,是为了让存档序列化稳定(EffectDef 只读、构造参数多)。</summary>
      public sealed class OpeningEffect
      {
          public string SourceCharId { get; set; }
          public Element Element { get; set; }
          public EffectKind Kind { get; set; }
          public int Value { get; set; }
          public int Turns { get; set; }
          public bool TargetAll { get; set; }
          public int BattlesLeft { get; set; }

          public OpeningEffect Clone() => (OpeningEffect)MemberwiseClone();
          public EffectDef ToEffect() => new EffectDef(Kind, Value, turns: Turns, targetAll: TargetAll);
      }

      /// <summary>同类开局效果只取最强(spec v6 §5.2 第 1 律):按 (Kind, TargetAll) 分组,
      /// Value 大者胜;Value 相同取 BattlesLeft 长者。</summary>
      public static class OpeningRules
      {
          public static List<OpeningEffect> Merge(IEnumerable<OpeningEffect> existing, IEnumerable<OpeningEffect> incoming) =>
              existing.Concat(incoming)
                  .Where(o => o.BattlesLeft > 0)
                  .GroupBy(o => (o.Kind, o.TargetAll))
                  .Select(g => g.OrderByDescending(o => o.Value).ThenByDescending(o => o.BattlesLeft).First().Clone())
                  .ToList();
      }
  }
  ```
